## 1. 系统概述

### 1.1 什么是动态快捷键

动态快捷键允许用户在浏览器扩展的 Popup 界面中自定义键盘快捷键，设置后立即在所有网页中生效。

**特点**:
- ✅ 用户可自定义（如 Ctrl+Shift+F）
- ✅ 实时同步到所有标签页
- ✅ 持久化存储
- ✅ 支持清除和重置

### 1.2 应用场景

| 功能 | 默认快捷键 | 可自定义 |
|------|-----------|---------|
| OCR 识别 | 无 | ✅ |
| 收藏文本 | Ctrl 键 | ✅ |

### 2.2 录制流程

**文件**: `popup/popup.js`

```javascript
// 步骤 1: 点击输入框，开始录制
favoritesShortcutInput.addEventListener('click', startRecording);

function startRecording() {
  // 防止重复录制
  if (isRecording) return;
  isRecording = true;

  // 更新 UI 状态
  input.classList.add('recording');
  input.value = '请按下快捷键组合...';
  input.disabled = true;

  // 监听键盘事件
  document.addEventListener('keydown', recordShortcut);
  document.addEventListener('keyup', finishRecording);
}

// 步骤 2: 记录按键
function recordShortcut(e) {
  e.preventDefault();

  // 获取修饰键
  const modifiers = [];
  if (e.ctrlKey) modifiers.push('Ctrl');
  if (e.altKey) modifiers.push('Alt');
  if (e.shiftKey) modifiers.push('Shift');
  if (e.metaKey) modifiers.push('Meta');

  // 验证：至少需要一个修饰键
  if (modifiers.length === 0) {
    input.value = '请至少按下一个修饰键';
    return;
  }

  // 构建快捷键字符串
  const mainKey = e.key;
  const shortcutString = [...modifiers, mainKey].join('+');

  // 实时显示
  input.value = shortcutString;
  // 如: "Ctrl+Shift+F"
}

// 步骤 3: 松开按键，完成录制
function finishRecording(e) {
  e.preventDefault();

  isRecording = false;
  input.classList.remove('recording');
  input.disabled = false;

  // 移除监听器
  document.removeEventListener('keydown', recordShortcut);
  document.removeEventListener('keyup', finishRecording);

  const shortcutString = input.value;

  // 验证格式
  if (!shortcutString || shortcutString.includes('请按下')) {
    input.value = '';
    return;
  }

  // 解析为对象
  const shortcut = parseShortcutString(shortcutString);

  // 保存
  saveShortcut(shortcut);
}
```

### 2.3 解析快捷键字符串

```javascript
function parseShortcutString(shortcutString) {
  // 输入: "Ctrl+Shift+F"
  const parts = shortcutString.split('+');
  // ['Ctrl', 'Shift', 'F']

  return {
    ctrlKey: parts.includes('Ctrl'),
    altKey: parts.includes('Alt'),
    shiftKey: parts.includes('Shift'),
    metaKey: parts.includes('Meta'),
    key: parts[parts.length - 1]  // 最后一个元素是主键
  };
}

// 输出示例:
// {
//   ctrlKey: true,
//   altKey: false,
//   shiftKey: true,
//   metaKey: false,
//   key: 'F'
// }
```

## 4. 同步机制

### 4.1 主动广播（推荐）

**Popup 端**:

```javascript
function saveShortcut(shortcut) {
  // 1. 保存到存储
  chrome.storage.local.set({ favoritesShortcut: shortcut });

  // 2. 主动广播给所有标签页
  chrome.tabs.query({}, (tabs) => {
    tabs.forEach(tab => {
      chrome.tabs.sendMessage(tab.id, {
        action: 'updateFavoritesShortcut',
        shortcut: shortcut
      }).catch(() => {
        // 忽略无法发送消息的标签页（如 chrome:// 页面）
      });
    });
  });
}
```

**Content Script 端**:

```javascript
chrome.runtime.onMessage.addListener((request, sender, sendResponse) => {
  if (request.action === 'updateFavoritesShortcut') {
    // 实时更新
    currentShortcut = request.shortcut;
    console.log('快捷键已更新:', currentShortcut);
    sendResponse({ success: true });
  }
});
```

### 4.2 被动监听（辅助）

**Content Script 端**:

```javascript
chrome.storage.onChanged.addListener((changes, areaName) => {
  if (areaName === 'local' && changes.favoritesShortcut) {
    const newShortcut = changes.favoritesShortcut.newValue;
    currentShortcut = newShortcut;
    console.log('检测到快捷键变化:', currentShortcut);
  }
});
```

## 5. 监听机制

### 5.1 全局键盘监听

**文件**: `content/content.js` 或 `content/content-ocr.js`

```javascript
// 当前快捷键状态
let currentShortcut = null;

// 监听键盘事件
document.addEventListener('keydown', (e) => {
  // 如果快捷键未设置，不处理
  if (!currentShortcut) return;

  // 检查是否匹配
  if (isShortcutMatch(e, currentShortcut)) {
    e.preventDefault();
    e.stopPropagation();

    // 执行功能
    triggerAction();
  }
});
```

### 5.2 匹配算法

```javascript
function isShortcutMatch(e, shortcut) {
  if (!shortcut) return false;

  return (
    e.ctrlKey === shortcut.ctrlKey &&
    e.altKey === shortcut.altKey &&
    e.shiftKey === shortcut.shiftKey &&
    e.metaKey === shortcut.metaKey &&
    e.key.toLowerCase() === shortcut.key.toLowerCase()
  );
}
```

### 5.3 匹配示例

```javascript
// 用户设置的快捷键
const shortcut = {
  ctrlKey: true,
  altKey: false,
  shiftKey: true,
  metaKey: false,
  key: 'F'
};

// 用户按下 Ctrl+Shift+F
const event = {
  ctrlKey: true,   // ✓ 匹配
  altKey: false,   // ✓ 匹配
  shiftKey: true,  // ✓ 匹配
  metaKey: false,  // ✓ 匹配
  key: 'F'         // ✓ 匹配
};

// isShortcutMatch(event, shortcut) → true
```

## 6. 完整示例

### 6.1 Popup 端完整代码

```javascript
// ===== 变量声明 =====
let isRecording = false;

// ===== DOM 元素 =====
const input = document.getElementById('favoritesShortcutInput');
const clearBtn = document.getElementById('clearFavoritesShortcutBtn');

// ===== 事件绑定 =====
input.addEventListener('click', startRecording);
clearBtn.addEventListener('click', clearShortcut);

// ===== 录制相关函数 =====
function startRecording() {
  if (isRecording) return;
  isRecording = true;

  input.classList.add('recording');
  input.value = '请按下快捷键组合...';
  input.disabled = true;

  document.addEventListener('keydown', recordShortcut);
  document.addEventListener('keyup', finishRecording);
}

function recordShortcut(e) {
  e.preventDefault();

  const modifiers = [];
  if (e.ctrlKey) modifiers.push('Ctrl');
  if (e.altKey) modifiers.push('Alt');
  if (e.shiftKey) modifiers.push('Shift');
  if (e.metaKey) modifiers.push('Meta');

  if (modifiers.length === 0) {
    input.value = '请至少按下一个修饰键';
    return;
  }

  const shortcutString = [...modifiers, e.key].join('+');
  input.value = shortcutString;
}

function finishRecording(e) {
  e.preventDefault();

  isRecording = false;
  input.classList.remove('recording');
  input.disabled = false;

  document.removeEventListener('keydown', recordShortcut);
  document.removeEventListener('keyup', finishRecording);

  const shortcutString = input.value;
  if (!shortcutString || shortcutString.includes('请按下')) {
    input.value = '';
    return;
  }

  const shortcut = parseShortcutString(shortcutString);
  saveShortcut(shortcut);
  input.value = formatShortcutDisplay(shortcutString);
}

// ===== 工具函数 =====
function parseShortcutString(str) {
  const parts = str.split('+');
  return {
    ctrlKey: parts.includes('Ctrl'),
    altKey: parts.includes('Alt'),
    shiftKey: parts.includes('Shift'),
    metaKey: parts.includes('Meta'),
    key: parts[parts.length - 1]
  };
}

function formatShortcutDisplay(str) {
  return str
    .replace('Control', 'Ctrl')
    .replace('Meta', 'Cmd');
}

// ===== 保存和清除 =====
function saveShortcut(shortcut) {
  chrome.storage.local.set({ favoritesShortcut: shortcut });

  // 广播给所有标签页
  chrome.tabs.query({}, (tabs) => {
    tabs.forEach(tab => {
      chrome.tabs.sendMessage(tab.id, {
        action: 'updateFavoritesShortcut',
        shortcut: shortcut
      }).catch(() => {});
    });
  });
}

function clearShortcut() {
  chrome.storage.local.remove('favoritesShortcut');
  input.value = '';

  // 通知所有标签页
  chrome.tabs.query({}, (tabs) => {
    tabs.forEach(tab => {
      chrome.tabs.sendMessage(tab.id, {
        action: 'clearFavoritesShortcut'
      }).catch(() => {});
    });
  });
}

// ===== 加载保存的快捷键 =====
function loadShortcut() {
  chrome.storage.local.get(['favoritesShortcut'], (result) => {
    if (result.favoritesShortcut) {
      const s = result.favoritesShortcut;
      const parts = [];
      if (s.ctrlKey) parts.push('Ctrl');
      if (s.altKey) parts.push('Alt');
      if (s.shiftKey) parts.push('Shift');
      if (s.metaKey) parts.push('Meta');
      parts.push(s.key);

      input.value = formatShortcutDisplay(parts.join('+'));
    }
  });
}

// 初始化
loadShortcut();
```

### 6.2 Content Script 端完整代码

```javascript
// ===== 当前快捷键 =====
let currentShortcut = null;

// ===== 加载快捷键 =====
function loadShortcut() {
  chrome.storage.local.get(['favoritesShortcut'], (result) => {
    if (result.favoritesShortcut) {
      currentShortcut = result.favoritesShortcut;
      console.log('快捷键已加载:', formatShortcut(currentShortcut));
    }
  });
}

// ===== 监听消息更新 =====
chrome.runtime.onMessage.addListener((request, sender, sendResponse) => {
  if (request.action === 'updateFavoritesShortcut') {
    currentShortcut = request.shortcut;
    console.log('快捷键已更新:', formatShortcut(currentShortcut));
    sendResponse({ success: true });
  } else if (request.action === 'clearFavoritesShortcut') {
    currentShortcut = null;
    console.log('快捷键已清除');
    sendResponse({ success: true });
  }
});

// ===== 监听 storage 变化 =====
chrome.storage.onChanged.addListener((changes, areaName) => {
  if (areaName === 'local' && changes.favoritesShortcut) {
    currentShortcut = changes.favoritesShortcut.newValue;
    console.log('检测到快捷键变化:', formatShortcut(currentShortcut));
  }
});

// ===== 监听键盘事件 =====
document.addEventListener('keydown', (e) => {
  if (!currentShortcut) return;

  if (isShortcutMatch(e, currentShortcut)) {
    e.preventDefault();
    e.stopPropagation();

    // 执行功能
    addToFavorites(getSelectedText(), window.location.href);
    showNotification('已收藏');
  }
});

// ===== 工具函数 =====
function isShortcutMatch(e, shortcut) {
  return (
    e.ctrlKey === shortcut.ctrlKey &&
    e.altKey === shortcut.altKey &&
    e.shiftKey === shortcut.shiftKey &&
    e.metaKey === shortcut.metaKey &&
    e.key.toLowerCase() === shortcut.key.toLowerCase()
  );
}

function formatShortcut(shortcut) {
  const parts = [];
  if (shortcut.ctrlKey) parts.push('Ctrl');
  if (shortcut.altKey) parts.push('Alt');
  if (shortcut.shiftKey) parts.push('Shift');
  if (shortcut.metaKey) parts.push('Meta');
  parts.push(shortcut.key);
  return parts.join('+');
}

function getSelectedText() {
  return window.getSelection().toString().trim();
}

// 初始化
loadShortcut();
```

## 总结

### 核心要点

1. **数据结构**: 使用对象描述快捷键 `{ ctrlKey, altKey, shiftKey, metaKey, key }`
2. **存储**: 使用 `chrome.storage.local` 持久化
3. **同步**: 主动广播 + 被动监听，双重保障
4. **监听**: 全局 `keydown` 事件 + 匹配算法

### 流程总结

```
用户设置 → 保存 Storage → 广播消息 → Content Script 更新 → 监听键盘 → 匹配触发
```

### 优势

- ✅ 用户可自定义
- ✅ 实时同步
- ✅ 持久化存储
- ✅ 跨标签页一致
